cmake_minimum_required(VERSION 3.22)
project(ninebot_native_tests LANGUAGES C CXX)
get_filename_component(MODULE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CORE "${MODULE}/src/main/cpp")
if(NOT DEFINED CAST_DEPS)
    get_filename_component(PROJECT_ROOT "${MODULE}/../.." ABSOLUTE)
    set(CAST_DEPS "${PROJECT_ROOT}/.deps")
endif()
if(DEFINED MBEDCRYPTO_LIBRARY)
    add_library(mbedcrypto STATIC IMPORTED)
    set_target_properties(mbedcrypto PROPERTIES IMPORTED_LOCATION "${MBEDCRYPTO_LIBRARY}")
else()
    set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${CAST_DEPS}/mbedtls" mbedtls)
endif()
add_executable(ninebot_native_tests core_tests.cpp "${CORE}/cast_protocol.cpp" "${CORE}/cast_core.cpp" "${CORE}/ninebot_gen3.cpp" "${CORE}/vendor/kcp/ikcp.c" "${CAST_DEPS}/cjson/cJSON.c")
target_compile_features(ninebot_native_tests PRIVATE cxx_std_17)
target_compile_definitions(ninebot_native_tests PRIVATE MZ_PREPARATION_FIXTURE="${CMAKE_CURRENT_LIST_DIR}/fixtures/mz-preparation-contract.json")
target_include_directories(ninebot_native_tests PRIVATE "${CORE}" "${CAST_DEPS}/cjson" "${CAST_DEPS}/mbedtls/include")
target_link_libraries(ninebot_native_tests PRIVATE mbedcrypto)
if(WIN32)
    target_link_libraries(ninebot_native_tests PRIVATE bcrypt)
endif()
enable_testing()
add_test(NAME ninebot_protocol_regression COMMAND ninebot_native_tests "${CMAKE_CURRENT_LIST_DIR}/fixtures/probe-vectors.json" "${CMAKE_CURRENT_LIST_DIR}/fixtures/v3-common-wire-validation.json")
